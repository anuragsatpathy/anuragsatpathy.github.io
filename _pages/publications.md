document.addEventListener('DOMContentLoaded', function () {
  var tabs = Array.prototype.slice.call(document.querySelectorAll('.vish-pub-tab'));
  var years = Array.prototype.slice.call(document.querySelectorAll('.vish-year'));
  var yearLinks = Array.prototype.slice.call(document.querySelectorAll('.vish-year-links a'));

  function applyFilter(filter) {
    tabs.forEach(function (tab) {
      var active = tab.getAttribute('data-filter') === filter;
      tab.classList.toggle('is-active', active);
      tab.setAttribute('aria-pressed', active ? 'true' : 'false');
    });

    years.forEach(function (yearSection) {
      var pubs = Array.prototype.slice.call(yearSection.querySelectorAll('.vish-pub'));
      var visibleCount = 0;

      pubs.forEach(function (pub) {
        var show = filter === 'all' || pub.getAttribute('data-category') === filter;
        pub.hidden = !show;
        if (show) visibleCount += 1;
      });

      yearSection.hidden = visibleCount === 0;
    });

    yearLinks.forEach(function (link) {
      var target = document.querySelector(link.getAttribute('href'));
      link.hidden = !target || target.hidden;
    });

    if (window.history && window.history.replaceState) {
      var suffix = filter === 'all' ? '' : '?type=' + encodeURIComponent(filter);
      window.history.replaceState(null, '', window.location.pathname + suffix);
    }
  }

  tabs.forEach(function (tab) {
    tab.addEventListener('click', function () {
      applyFilter(tab.getAttribute('data-filter'));
    });
  });

  var params = new URLSearchParams(window.location.search);
  var initial = params.get('type');
  var valid = ['journal', 'conference', 'book', 'poster'];
  applyFilter(valid.indexOf(initial) !== -1 ? initial : 'all');
});
</script>
